34// ── FFT Parameters ─────────────────────────────────────────────────────────────
35// HR_FFT_SAMPLES must be a power of 2.
36// At HR_SAMPLE_RATE = 100 Hz → 512 samples = 5.12 s window.
37// Frequency resolution = HR_SAMPLE_RATE / HR_FFT_SAMPLES = 0.195 Hz ≈ 11.7 BPM/bin.
38// Parabolic interpolation refines this to < 3 BPM accuracy in practice.
39constexpr uint16_t HR_FFT_SAMPLES = 512;
40constexpr float HR_SAMPLE_RATE = 100.0f; // Hz — must match MAX30105_RATE_HZ
41constexpr uint16_t HR_FFT_UPDATE = 64; // Recompute every N new samples
42 // (= 0.64 s sliding step at 100 Hz)
43
44// ── Detection Range ────────────────────────────────────────────────────────────
45constexpr float HR_MIN_BPM = 40.0f;
46constexpr float HR_MAX_BPM = 200.0f;

48// ── Finger Detection Threshold ─────────────────────────────────────────────────
49constexpr long HR_FINGER_THRESHOLD = 50000;

51// ── Bandpass IIR Filter Coefficients ──────────────────────────────────────────
52// Cascaded 1st-order high-pass (0.5 Hz) + 1st-order low-pass (4 Hz) at 100 Hz.
53//
54// High-pass: α_hp = RC/(RC+dt), RC = 1/(2π·0.5) = 0.31831 s, dt = 0.01 s
55// α_hp = 0.31831 / (0.31831 + 0.01) = 0.96952
56//
57// Low-pass: α_lp = dt/(RC+dt), RC = 1/(2π·4.0) = 0.03979 s, dt = 0.01 s
58// α_lp = 0.01 / (0.03979 + 0.01) = 0.20083
59constexpr float HR_IIR_HP_ALPHA = 0.96952f; // high-pass pole (removes DC / <0.5 Hz)
60constexpr float HR_IIR_LP_ALPHA = 0.20083f; // low-pass pole (removes >4 Hz noise)
61
62// ── SNR Quality Gate ───────────────────────────────────────────────────────────
63// Minimum ratio of peak-bin magnitude to average noise-floor magnitude.
64// Results that fall below this threshold are discarded as unreliable.
65constexpr float HR_SNR_MIN_RATIO = 2.5f;
66
67// ── Median Filter ──────────────────────────────────────────────────────────────
68// The last HR_MEDIAN_SIZE valid FFT BPM readings are stored in a ring buffer.
69// Their median is computed and fed into the EMA, making the output immune to
70// bin-hopping outliers (even if 2 of 5 readings land on the wrong bin, the
71// median still selects the correct one).
72constexpr uint8_t HR_MEDIAN_SIZE = 5;
73
74// ── EMA Smoothing ──────────────────────────────────────────────────────────────
75// Exponential moving average weight applied to each new FFT BPM estimate.
76// Lower α → smoother (slower to track changes); higher α → more responsive.
77// 0.15 gives good stability at the ~0.64 s update rate on top of the median filter.
78constexpr float HR_EMA_ALPHA = 0.15f;
